Create table Teacher_46S(
	Tid Number(4) Primary key,
        Name Varchar2(15) NOT NULL,
	 Dept Varchar2(15) NOT NULL
);


Create table Subject_46S(
	
	Subno Number(4) Primary Key ,
	 Subtitle Varchar2(15) NOT NULL
);

Create table TaughtBy_46S(
	Tid Number(4),
	Subno Number(4),
	primary key(Tid,Subno),
	Foreign key(Tid) references Teacher_46S(Tid),
	Foreign key(Subno) references Subject_46S(Subno)
);

Create table Student_46S(
	Rollno Number(3) Primary key,
	Sname Varchar2(15) NOT NULL,
	City Varchar2(15) NOT NULL
);
