Create table ZIPCODES46S(
	ZIP Number(6) PRIMARY KEY,
	CITY Varchar2(15)
);


Create table EMPLOYEES46S(
	ENO Number(4) Primary Key,
	ENAME Varchar2(15) NOT NULL,
	ZIP Number(6),
	HDATE Date NOT NULL,
	Foreign key(ZIP) references ZIPCODES46S(ZIP)
);

Create table PARTS46S(
	PNO Number(4) Primary Key,
	PNAME Varchar2(15) NOT NULL,
	QOH Number(5) NOT NULL,
	PRICE Number(5) NOT NULL,
	LEV Number(3)
);


Create table CUSTOMERS46S (
 	CNO number(4) Primary Key,
 	CNAME Varchar2(15) NOT NULL,
 	STREET Varchar2(10),
 	ZIP Number(6) references ZIPCODES46S(ZIP),
 	PHONE Number(10)
);


CREATE TABLE ORDERS46S (
    ONO NUMBER(4) PRIMARY KEY,
    CNO NUMBER(4),
    ENO NUMBER(4),
    RECEIVED DATE,
    SHIPPED DATE,
    FOREIGN KEY (CNO) REFERENCES CUSTOMERS46S(CNO),
    FOREIGN KEY (ENO) REFERENCES EMPLOYEES46S(ENO)
);


CREATE TABLE ODETAILS46S (
    ONO NUMBER(4),
    PNO NUMBER(4),
    QTY NUMBER(5),
    PRIMARY KEY (ONO, PNO),
    FOREIGN KEY (ONO) REFERENCES ORDERS46S(ONO),
    FOREIGN KEY (PNO) REFERENCES PARTS46S(PNO)
);
